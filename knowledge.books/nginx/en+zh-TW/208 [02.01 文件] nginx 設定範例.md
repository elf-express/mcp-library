## Пример конфигурации nginx｜nginx 設定範例

Пример конфигурации сайта, который передаёт все запросы бэкенду, кроме картинок и запросов, начинающихся с "/download/".

以下是一個網站配置範例，它將所有請求都傳遞給後端，但圖像請求和以“/download/”開頭的請求除外。

> ```
> user  www www;
>
> 使用者 www www;
> 
> worker_processes  2;
>
> 工作進程 2；
> 
> pid /var/run/nginx.pid;
> 
> #                          [ debug | info | notice | warn | error | crit ]｜[調試 | 資訊 | 通知 | 警告 | 錯誤 | 嚴重錯誤]
> 
> error_log  /var/log/nginx.error_log  info;
>
> 錯誤日誌 /var/log/nginx.error_log 訊息;
> 
> events {
>
> 事件 {
>     worker_connections   2000;
>
>     worker_connections 2000;
> 
>     # use [ kqueue | epoll | /dev/poll | select | poll ];｜使用 [ kqueue | epoll | /dev/poll | select | poll ]；
>     use kqueue;
>
>     使用 kqueue；
> }
> 
> http {
> 
>     include       conf/mime.types;
>
>     包含 conf/mime.types；
>     default_type  application/octet-stream;
>
>     預設類型 application/octet-stream;
> 
> 
>     log_format main      '$remote_addr - $remote_user [$time_local] '
>
>     log_format main '$remote_addr - $remote_user [$time_local] '
>                          '"$request" $status $bytes_sent '
>                          '"$http_referer" "$http_user_agent" '
>                          '"$gzip_ratio"';
> 
>     log_format download  '$remote_addr - $remote_user [$time_local] '
>
>     log_format 下載 '$remote_addr - $remote_user [$time_local] '
>                          '"$request" $status $bytes_sent '
>                          '"$http_referer" "$http_user_agent" '
>                          '"$http_range" "$sent_http_content_range"';
> 
>     client_header_timeout  3m;
>
>     client_header_timeout 3m;
>     client_body_timeout    3m;
>
>     client_body_timeout 3m;
>     send_timeout           3m;
>
>     發送超時時間為 3 分鐘；
> 
>     client_header_buffer_size    1k;
>
>     client_header_buffer_size 1k;
>     large_client_header_buffers  4 4k;
>
>     large_client_header_buffers 4 4k;
> 
>     gzip on;
>
>     啟用 gzip 壓縮；
>     gzip_min_length  1100;
>
>     gzip_min_length 1100;
>     gzip_buffers     4 8k;
>
>     gzip_buffers 4 8k;
>     gzip_types       text/plain;
>
>     gzip_types text/plain;
> 
>     output_buffers   1 32k;
>
>     輸出緩衝區 1 32k；
>     postpone_output  1460;
>
>     推遲輸出 1460；
> 
>     sendfile         on;
>
>     發送文件已開啟；
>     tcp_nopush       on;
>
>     tcp_nopush 已開啟；
>     tcp_nodelay      on;
>
>     tcp_nodelay 開啟；
>     send_lowat       12000;
>
>     send_lowat 12000;
> 
>     keepalive_timeout  75 20;
>
>     keepalive_timeout 75 20;
> 
>     #lingering_time     30;
>
>     #lingering_time 30;
>     #lingering_timeout  10;
>
>     #lingering_timeout 10;
>     #reset_timedout_connection  on;
>
>     #reset_timedout_connection on;
> 
> 
>     server {
>
>     伺服器 {
>         listen        one.example.com;
>
>         收聽 one.example.com；
>         server_name   one.example.com  www.one.example.com;
>
>         server_name one.example.com www.one.example.com;
> 
>         access_log   /var/log/nginx.access_log  main;
>
>         access_log /var/log/nginx.access_log main;
> 
>         location / {
>
>         地點 / {
>             proxy_pass         http://127.0.0.1/;
>
>             proxy_pass http://127.0.0.1/;
>             proxy_redirect     off;
>
>             代理重定向關閉；
> 
>             proxy_set_header   Host             $host;
>
>             proxy_set_header Host $host;
>             proxy_set_header   X-Real-IP        $remote_addr;
>
>             proxy_set_header X-Real- IP $remote_addr;
>             #proxy_set_header  X-Forwarded-For  $proxy_add_x_forwarded_for;
>
>             #proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
> 
>             client_max_body_size       10m;
>
>             client_max_body_size 10m;
>             client_body_buffer_size    128k;
>
>             client_body_buffer_size 128k;
> 
>             client_body_temp_path      /var/nginx/client_body_temp;
>
>             client_body_temp_path /var/nginx/client_body_temp;
> 
>             proxy_connect_timeout      70;
>
>             proxy_connect_timeout 70;
>             proxy_send_timeout         90;
>
>             proxy_send_timeout 90;
>             proxy_read_timeout         90;
>
>             proxy_read_timeout 90;
>             proxy_send_lowat           12000;
>
>             proxy_send_lowat 12000;
> 
>             proxy_buffer_size          4k;
>
>             proxy_buffer_size 4k;
>             proxy_buffers              4 32k;
>
>             proxy_buffers 4 32k;
>             proxy_busy_buffers_size    64k;
>
>             proxy_busy_buffers_size 64k;
>             proxy_temp_file_write_size 64k;
> 
>             proxy_temp_path            /var/nginx/proxy_temp;
>
>             proxy_temp_path /var/nginx/proxy_temp;
> 
>             charset  koi8-r;
>
>             charset koi8-r;
>         }
> 
>         error_page  404  /404.html;
>
>         錯誤頁 404 /404.html;
> 
>         location = /404.html {
>             root  /spool/www;
>
>             根目錄 /spool/www;
>         }
> 
>         location /old_stuff/ {
>             rewrite   ^/old_stuff/(.*)$  /new_stuff/$1  permanent;
>
>             重寫 ^/old_stuff/(.*)$ /new_stuff/$1 永久；
>         }
> 
>         location /download/ {
> 
>             valid_referers  none  blocked  server_names  *.example.com;
>
>             valid_referers none blocked server_names *.example.com;
> 
>             if ($invalid_referer) {
>
>             如果 ($invalid_referer) {
>                 #rewrite   ^/   http://www.example.com/;
>
>                 #重寫 ^/ http://www.example.com/;
>                 return   403;
>
>                 返回 403；
>             }
> 
>             #rewrite_log  on;
>
>             #rewrite_log on;
> 
>             # rewrite /download/*/mp3/*.any_ext to /download/*/mp3/*.mp3｜將 /download/*/mp3/*.any_ext 改寫為 /download/*/mp3/*.mp3
>             rewrite ^/(download/.*)/mp3/(.*)\..*$
>
>             重寫 ^/(download/.*)/mp3/(.*)\..*$
>                     /$1/mp3/$2.mp3                   break;
>
>                     /$1/mp3/$2.mp3 break;
> 
>             root         /spool/www;
>
>             根目錄 /spool/www;
>             #autoindex    on;
>
>             #自動索引開啟；
>             access_log   /var/log/nginx-download.access_log  download;
>
>             access_log /var/log/nginx-download.access_log 下載;
>         }
> 
>         location ~* \.(jpg|jpeg|gif)$ {
>             root         /spool/www;
>
>             根目錄 /spool/www;
>             access_log   off;
>
>             關閉訪問日誌；
>             expires      30d;
>
>             有效期限30天；
>         }
>     }
> }
> ```
