## Example nginx configuration｜nginx 設定範例

An example site configuration that passes all requests to the backend except images and requests starting with "/download/".

範例網站配置，將除圖像和以“/download/”開頭的請求之外的所有請求傳遞到後端。

> ```
> user  www www;
>
> 使用者 www www;
> 
> worker_processes  2;
>
> 工人進程2；
> 
> pid /var/run/nginx.pid;
> 
> #                          [ debug | info | notice | warn | error | crit ]｜[ 調試 |訊息 |通知 |警告|錯誤|暴擊]
> 
> error_log  /var/log/nginx.error_log  info;
>
> error_log /var/log/nginx.error_log 訊息；
> 
> events {
>
> 事件{
>     worker_connections   2000;
>
>     工人連接2000；
> 
>     # use [ kqueue | epoll | /dev/poll | select | poll ];｜使用 [ kqueue |電子輪詢 | /dev/民意調查 |選擇 |民意調查]；

> 
> http {
>
> http{
> 
>     include       conf/mime.types;

>     包括conf/mime.types；

>

> 
> 
>     log_format main      '$remote_addr - $remote_user [$time_local] '
>
>     log_format main '$remote_addr - $remote_user [$time_local] '
>                          '"$request" $status $bytes_sent '
>
>                          '“$request”$status $bytes_sent'
>                          '"$http_referer" "$http_user_agent" '
>
>                          '“$http_referer”“$http_user_agent”'
>                          '"$gzip_ratio"';
>
>                          '“$gzip_ratio”';
> 
>     log_format download  '$remote_addr - $remote_user [$time_local] '
>
>     log_format download '$remote_addr - $remote_user [$time_local] '
>                          '"$request" $status $bytes_sent '
>
>                          '“$request”$status $bytes_sent'
>                          '"$http_referer" "$http_user_agent" '
>
>                          '“$http_referer”“$http_user_agent”'
>                          '"$http_range" "$sent_http_content_range"';
>
>                          '“$http_range”“$sent_http_content_range”';
> 
>     client_header_timeout  3m;
>
>     client_header_timeout 3m；
>     client_body_timeout    3m;
>
>     client_body_timeout 3m；
>     send_timeout           3m;

>     發送超時3m；

>     client_header_buffer_size    1k;

>     client_header_buffer_size 1k；

>     gzip 開啟；

>

>

> 
>     output_buffers   1 32k;
>

>     postpone_output  1460;
>
>     [1460] 延後_輸出
> 
>     sendfile         on;
>
>     發送文件；
>     tcp_nopush       on;
>
>     tcp_nopush 開啟；
>     tcp_nodelay      on;
>
>     tcp_nodelay 開啟；
>     send_lowat       12000;
>
>     send_lowat 12000；
> 
>     keepalive_timeout  75 20;
>

>     #lingering_time     30;
>

>     #lingering_timeout  10;

>     #lingering_timeout 10；
>     #reset_timedout_connection  on;
>
>     #reset_timedout_connection on；
> 
> 
>     server {
>
>     伺服器{
>         listen        one.example.com;
>
>         聽 one.example.com；
>         server_name   one.example.com  www.one.example.com;
>
>         伺服器名稱 one.example.com www.one.example.com；

>         access_log   /var/log/nginx.access_log  main;

>         access_log /var/log/nginx.access_log main;

>             client_max_body_size       10m;
>
>             client_max_body_size 10m；
>             client_body_buffer_size    128k;
>
>             client_body_buffer_size 128k；

>             client_body_temp_path      /var/nginx/client_body_temp;
>
>             client_body_temp_path /var/nginx/client_body_temp;
> 
>             proxy_connect_timeout      70;
>
>             proxy_connect_timeout 70；
>             proxy_send_timeout         90;
>
>             proxy_send_timeout 90；
>             proxy_read_timeout         90;
>
>             proxy_read_timeout 90；
>             proxy_send_lowat           12000;
>
>             proxy_send_lowat 12000；
> 
>             proxy_buffer_size          4k;
>
>             proxy_buffer_size 4k；

>

> 
>             charset  koi8-r;
>
>             字符集 koi8-r；
>         }
> 
>         error_page  404  /404.html;
>
>         error_page 404 /404.html;
> 
>         location = /404.html {
>
>         位置= /404.html {
>             root  /spool/www;
>
>             根/spool/www；
>         }
> 
>         location /old_stuff/ {
>
>         位置 /old_stuff/ {
>             rewrite   ^/old_stuff/(.*)$  /new_stuff/$1  permanent;
>
>             重寫 ^/old_stuff/(.*)$ /new_stuff/$1 永久；
>         }
> 
>         location /download/ {
>
>         位置/下載/ {
> 
>             valid_referers  none  blocked  server_names  *.example.com;
>
>             valid_referers 沒有阻止 server_names *.example.com；
> 
>             if ($invalid_referer) {
>
>             如果（$invalid_referer）{
>                 #rewrite   ^/   http://www.example.com/;
>
>                 #重寫^/http://www.example.com/;
>                 return   403;
>
>                 返回403；
>             }
> 
>             #rewrite_log  on;
>
>             #rewrite_log on；
>

>                     /$1/mp3/$2.mp3                   break;
>
>                     /$1/mp3/$2.mp3 中斷；
> 
>             root         /spool/www;
>
>             根/spool/www；
>             #autoindex    on;
>
>             #自動索引；
>             access_log   /var/log/nginx-download.access_log  download;

>             access_log /var/log/nginx-download.access_log 下載；

>

>

>             expires      30d;
>

>         }
